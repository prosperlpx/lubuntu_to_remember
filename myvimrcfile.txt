" ==========================================
" Modern Defaults & UI
" ==========================================
syntax on                    " Enable syntax highlighting
filetype plugin indent on    " Enable filetype detection, plugins, and indents

set number                   " Show line numbers
set relativenumber           " Show relative line numbers for easier jumping
set mouse=a                  " Enable mouse support in all modes
set clipboard=unnamedplus    " Use system clipboard
set termguicolors            " Enable 24-bit RGB colors

" Apply the theme we downloaded locally
colorscheme onedark

" ==========================================
" Tabs, Indentation & Search
" ==========================================
set tabstop=4                " Number of visual spaces per TAB
set softtabstop=4            " Number of spaces a TAB counts for while editing
set shiftwidth=4             " Number of spaces to use for autoindent
set expandtab                " Convert TABs to spaces
set smartindent              " Smart auto-indenting for new lines

set incsearch                " Incremental search (highlights as you type)
set hlsearch                 " Highlight search results
set ignorecase               " Ignore case when searching
set smartcase                " ...unless the query contains capital letters

" ==========================================
" Built-in Offline Intellisense
" ==========================================
" Omnifunc gives you language-aware completion purely using built-in Vim logic
set omnifunc=syntaxcomplete#Complete

" Optimize completion menu behavior
set completeopt=menu,menuone,noselect,noinsert

" ==========================================
" Local Plugin Keybindings
" ==========================================
" Toggle File Explorer Sidebar with Ctrl+n
nnoremap <C-n> :NERDTreeToggle<CR>
